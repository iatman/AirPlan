{{/* Render the initial calendar at build time; JavaScript enhances year selection and filtering. */}} {{- $year := int .year -}} {{- $first := time (printf "%04d-01-01T00:00:00Z" $year) -}} {{- $last := time (printf "%04d-12-31T00:00:00Z" $year) -}} {{- $start := $first.AddDate 0 0 (int (mul -1 (int $first.Weekday))) -}} {{- $days := add 1 (div (sub $last.Unix $start.Unix) 86400) -}} {{- $weeks := int (div (add $days 6) 7) -}} {{- $records := newScratch -}} {{- range .pages -}} {{- $date := .Date.Format "2006-01-02" -}} {{- $daily := $records.Get $date | default slice -}} {{- $records.Set $date ($daily | append .) -}} {{- end -}}
{{- range seq 0 11 -}} {{- $month := $first.AddDate 0 (int .) 0 -}} {{- $week := int (div (div (sub $month.Unix $start.Unix) 86400) 7) -}} {{ add . 1 }}月 {{- end -}} {{- range slice 1 3 5 -}}{{ index (slice "日" "一" "二" "三" "四" "五" "六") . }}{{- end -}} {{- range seq 0 (sub (mul $weeks 7) 1) -}} {{- $day := $start.AddDate 0 0 (int .) -}} {{- $date := $day.Format "2006-01-02" -}} {{- $inside := eq $day.Year $year -}} {{- $daily := $records.Get $date | default slice -}} {{- $count := len $daily -}} {{- $column := add 2 (int (div . 7)) -}} {{- $row := add 2 (mod . 7) -}} {{- if and $inside $count -}} {{- $article := index $daily 0 -}} {{- else -}} {{- end -}} {{- end -}}
